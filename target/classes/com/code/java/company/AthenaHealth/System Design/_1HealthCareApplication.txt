Low-Level Design (LLD) for Patients and Appointments RESTful API
// Tech stack: Java, Spring Boot, JPA (Hibernate), H2/PostgreSQL


// ========== 1. Patient Entity ==========
@Entity
public class Patient{
    @Id @GeneratedValue
    private Long id;
    private String name;
    private String email;
    private String phone;
    private LocalDate dob;

    //getters & setters
}

// ========== 2. Appointment Entity ==========
@Entity
public class Appointment {
    @Id @GeneratedValue
    private Long id;

    @ManyToOne
    private Patient patient;

    private Long doctorId;
    private LocalDateTime appointmentTime;
    private String status; // BOOKED, CANCELLED
    // Getters/Setters
}

// ========== 3. Repositories ==========
@Repository
public interface PatientRepository extends JpaRepository<Patient, Long> {}
@Repository
public interface AppointmentRepository extends JpaRepository<Appointment, Long> {
    boolean existsByDoctorIdAndAppointmentTime(Long doctorId, LocalDateTime time);
    List<Appointment> findByPatientId(Long patientId);
}

// ========== 4. DTOs ==========
public class AppointmentRequest {
    private Long patientId;
    private Long doctorId;
    private LocalDateTime appointmentTime;
}

public class AppointmentResponse {
    private Long appointmentId;
    private String status;
    private String message;
    // Constructors/Getters/Setters
}
// ========== 5. Service ==========
@Service
public class AppointmentService {
    @Autowired
    private AppointmentRepository appointmentRepo;
    @Autowired
    private PatientRepository patientRepo;

    public AppointmentResponse bookAppointment(AppointmentRequest req) {
        if (appointmentRepo.existsByDoctorIdAndAppointmentTime(req.getDoctorId(), req.getAppointmentTime())) {
            return new AppointmentResponse(null, "FAILED", "Slot already booked");
        }
        Patient patient = patientRepo.findById(req.getPatientId()).orElseThrow();
        Appointment appt = new Appointment();
        appt.setPatient(patient);
        appt.setDoctorId(req.getDoctorId());
        appt.setAppointmentTime(req.getAppointmentTime());
        appt.setStatus("BOOKED");
        appt = appointmentRepo.save(appt);
        return new AppointmentResponse(appt.getId(), appt.getStatus(), "Booked successfully");
    }

    public List<Appointment> getAppointmentsForPatient(Long patientId) {
        return appointmentRepo.findByPatientId(patientId);
    }
}

// ========== 6. Controller ==========
@RestController
@RequestMapping("/api")
public class AppointmentController {
    @Autowired
    private AppointmentService appointmentService;

    @PostMapping("/appointments")
    public ResponseEntity<AppointmentResponse> book(@RequestBody AppointmentRequest request) {
        AppointmentResponse res = appointmentService.bookAppointment(request);
        return res.getStatus().equals("FAILED")
            ? ResponseEntity.status(HttpStatus.CONFLICT).body(res)
            : ResponseEntity.ok(res);
    }

    @GetMapping("/patients/{id}/appointments")
    public List<Appointment> getAppointments(@PathVariable Long id) {
        return appointmentService.getAppointmentsForPatient(id);
    }
}
